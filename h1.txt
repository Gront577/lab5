#!/bin/bash

echo 'Time, MEM_used, Swap_used, UIRT_1, RES_1, %MEM_1, UIRT_2, RES_2, %MEM_2' > mem2.log
./mem.bash &
pid1=$!
./mem2.bash &
pid2=$!

while kill -0 "$pid1" 2>/dev/null || kill -0 "$pid2" 2>/dev/null
do
    memUsed=$(free -m | awk '/Mem:/ {print $3}')
    swapUsed=$(free -m | awk '/Swap:/ {print $3}')

    process=$(top -b -n 1 | awk -v p1="$pid1" -v p2="$pid2" '
        $1 == p1 {virt1=$5; res1=$6; mem1=$10}
        $1 == p2 {virt2=$5; res2=$6; mem2=$10}
        END {printf "%s: %s: %s: %s: %s: %s", virt1, res1, mem1, virt2, res2, mem2}')

    echo "$(date '+%H:%M:%S'), $memUsed, $swapUsed, $process" >> mem2.log
    sleep 5
done

echo "Finished"
