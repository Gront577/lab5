#!/bin/bash

echo "Timestamp,UsedMem(MB),UsedSwap(MB),VIRT(MB),RES(MB),%MEM" > memory_usage.log

./mem.bash &
echo "Мониторинг PID: $!"

while kill -0 "$!" 2>/dev/null; do
    # Получаем timestamp
    timestamp=$(date +%s)
    
    # Системная память
    mem_used=$(free -m | awk '/Mem:/{print $3}')
    swap_used=$(free -m | awk '/Swap:/{print $3}')
    
    # Данные процесса
    process_data=$(top -b -n 1 -p "$!" | awk -v pid="$!" '
        $1 == pid {
            printf "%.1f,%.1f,%.1f", $5, $6, $10
        }
    ')
    
    # Запись данных
    echo "$timestamp,$mem_used,$swap_used,$process_data" >> memory_usage.log
    
    sleep 1
done
