#!/bin/bash

# Очистка лог-файла и добавление заголовков
echo "Time,UsedMem(MB),UsedSwap(MB),VIRT(MB),RES(MB),Process%" > memory_usage.log

# Запуск mem.bash в фоне
./mem.bash &
echo "Мониторинг памяти процесса mem.bash (PID: $!)"

while kill -0 "$!" 2>/dev/null; do
    # Системная память
    mem_used=$(free -m | awk '/Mem:/{print $3}')
    swap_used=$(free -m | awk '/Swap:/{print $3}')
    
    # Данные процесса
    process_data=$(top -b -n 1 -p "$!" | awk -v pid="$!" '
        $1 == pid {
            printf "%.1f, %.1f, %.1f", $5, $6, $10  # VIRT(MB), RES(MB), %MEM
        }
    ')
    
    # Запись данных
    echo "$(date '+%H:%M:%S'),$mem_used,$swap_used,$process_data" >> memory_usage.log
    
    sleep 1
done

echo "Мониторинг завершен. Данные сохранены в memory_usage.log"
